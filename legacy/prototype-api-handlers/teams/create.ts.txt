// Get existing team if it already exists for this user
async function getTeamForUser(userId: string): Promise<{ id: string; name: string; slug: string } | null> {
  const [team] = await db
    .select({ id: team.id, name: team.name, slug: team.slug })
    .from(team)
    .where(eq(team.userId, userId))
    .limit(1)
    .returning()
  return team
}
